  <script>
    // Navigation Routing
    function switchPage(pageId) {
      document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
      document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));
      document.getElementById(`page-${pageId}`).classList.add('active');
      event.target.classList.add('active');

      if (pageId === 'reviewSurveys') {
        renderReviewTable();
      }
    }

    // Capture Geolocation
    function captureGPS() {
      if ("geolocation" in navigator) {
        navigator.geolocation.getCurrentPosition(pos => {
          document.getElementById('gpsCoords').value = `${pos.coords.latitude.toFixed(5)}, ${pos.coords.longitude.toFixed(5)}`;
        }, err => {
          alert("Error getting location: " + err.message);
        });
      } else {
        alert("Geolocation is not supported by your browser.");
      }
    }

    // Dynamic Load Table
    function addLoadRow(item = '', qty = 1, watts = 0) {
      const tbody = document.querySelector('#loadTable tbody');
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td><input type="text" class="load-item" value="${item}" required style="width: 100%;"></td>
        <td><input type="number" class="load-qty" value="${qty}" min="1" onchange="calculateTotal()" style="width: 100%;"></td>
        <td><input type="number" class="load-watts" value="${watts}" min="0" onchange="calculateTotal()" style="width: 100%;"></td>
        <td class="row-total">${qty * watts} W</td>
        <td><button type="button" class="btn btn-danger" onclick="this.closest('tr').remove(); calculateTotal();">X</button></td>
      `;
      tbody.appendChild(tr);
      calculateTotal();
    }

    function calculateTotal() {
      let grandTotal = 0;
      document.querySelectorAll('#loadTable tbody tr').forEach(row => {
        const qty = parseFloat(row.querySelector('.load-qty').value) || 0;
        const watts = parseFloat(row.querySelector('.load-watts').value) || 0;
        const total = qty * watts;
        row.querySelector('.row-total').textContent = `${total} W`;
        grandTotal += total;
      });
      document.getElementById('totalLoadDisplay').textContent = grandTotal;
      return grandTotal;
    }

    // Save Survey to LocalStorage
    document.getElementById('surveyForm').addEventListener('submit', function(e) {
      e.preventDefault();

      const items = [];
      document.querySelectorAll('#loadTable tbody tr').forEach(row => {
        items.push({
          item: row.querySelector('.load-item').value,
          qty: row.querySelector('.load-qty').value,
          watts: row.querySelector('.load-watts').value
        });
      });

      const surveyData = {
        id: Date.now(),
        clientName: document.getElementById('clientName').value,
        siteLocation: document.getElementById('siteLocation').value,
        surveyorName: document.getElementById('surveyorName').value,
        surveyDate: document.getElementById('surveyDate').value,
        gpsCoords: document.getElementById('gpsCoords').value,
        notes: document.getElementById('siteNotes').value,
        totalLoad: calculateTotal(),
        items: items
      };

      const existing = JSON.parse(localStorage.getItem('siteSurveys') || '[]');
      existing.push(surveyData);
      localStorage.setItem('siteSurveys', JSON.stringify(existing));

      alert('Survey saved successfully!');
      this.reset();
      document.querySelector('#loadTable tbody').innerHTML = '';
      calculateTotal();
    });

    // Render Review List
    function renderReviewTable() {
      const surveys = JSON.parse(localStorage.getItem('siteSurveys') || '[]');
      const tbody = document.querySelector('#reviewTable tbody');
      tbody.innerHTML = '';

      surveys.forEach(s => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td>${s.surveyDate}</td>
          <td>${s.clientName}</td>
          <td>${s.siteLocation}</td>
          <td>${s.surveyorName}</td>
          <td>${s.totalLoad} W</td>
          <td>
            <button class="btn btn-danger" onclick="deleteSurvey(${s.id})">Delete</button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }

    function deleteSurvey(id) {
      let surveys = JSON.parse(localStorage.getItem('siteSurveys') || '[]');
      surveys = surveys.filter(s => s.id !== id);
      localStorage.setItem('siteSurveys', JSON.stringify(surveys));
      renderReviewTable();
    }

    // Initialize Default State
    document.addEventListener('DOMContentLoaded', () => {
      document.getElementById('surveyDate').valueAsDate = new Date();
      addLoadRow('Lighting Circuit', 1, 100);
    });

    // Register PWA Service Worker
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('./sw.js').catch(err => console.log('SW Registration failed:', err));
    }
